local function N(cls, props, par)
	local i = Instance.new(cls)
	for k, v in pairs(props) do i[k] = v end
	i.Parent = par
	return i
end

local function round(p)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(1, 0)
	c.Parent = p
	return c
end

local function getFoot(ch)
	return ch:FindFirstChild("LeftFoot") or ch:FindFirstChild("Left Foot")
		or ch:FindFirstChild("RightFoot") or ch:FindFirstChild("Right Foot")
		or ch:FindFirstChild("LeftLeg") or ch:FindFirstChild("Left Leg")
		or ch:FindFirstChild("HumanoidRootPart")
end

local function beamProps(a0, a1, w, tr)
	return { Attachment0 = a0, Attachment1 = a1, Width0 = w, Width1 = w, FaceCamera = true,
		LightEmission = 1, LightInfluence = 0, ZOffset = CFG.XRAY_Z,
		Color = ColorSequence.new(rainbowColor), Transparency = NumberSequence.new(tr), Enabled = false }
end

local losParams = RaycastParams.new()
losParams.FilterType = Enum.RaycastFilterType.Exclude
losParams.IgnoreWater = true

local function hasLineOfSight(from, part, ch)
	local filter = {}
	if LP.Character then filter[#filter + 1] = LP.Character end
	if ch then filter[#filter + 1] = ch end
	losParams.FilterDescendantsInstances = filter
	local hit = workspace:Raycast(from, part.Position - from, losParams)
	return hit == nil
end

local function keepOnScreen(frame, w, h)
	local cam = workspace.CurrentCamera
	if not cam then return end
	local vp = cam.ViewportSize
	local abs = frame.AbsoluteSize
	local fw = w or abs.X
	local fh = h or abs.Y
	if fw <= 0 or fh <= 0 then return end
	local margin = 4
	local px = frame.Position.X.Scale * vp.X + frame.Position.X.Offset
	local py = frame.Position.Y.Scale * vp.Y + frame.Position.Y.Offset
	local maxX = math.max(margin, vp.X - fw - margin)
	local maxY = math.max(margin, vp.Y - fh - margin)
	frame.Position = UDim2.fromOffset(math.clamp(px, margin, maxX), math.clamp(py, margin, maxY))
end

local function makeDrag(handle, frame)
	local on = false
	local p0, f0
	handle.InputBegan:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			on = true
			p0 = i.Position
			f0 = frame.Position
		end
	end)
	UIS.InputChanged:Connect(function(i)
		if on and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
			local d = i.Position - p0
			frame.Position = UDim2.new(f0.X.Scale, f0.X.Offset + d.X, f0.Y.Scale, f0.Y.Offset + d.Y)
			keepOnScreen(frame)
		end
	end)
	UIS.InputEnded:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			on = false
		end
	end)
end

local function toggle(key, default, cb)
	order = order + 1
	local row = N("Frame", { Size = UDim2.new(1, 0, 0, 28), BackgroundTransparency = 1, ZIndex = 2, LayoutOrder = order }, scroll)
	local lbl = N("TextLabel", { Size = UDim2.new(1, -50, 1, 0), BackgroundTransparency = 1, Text = T(key), TextColor3 = Color3.fromRGB(210, 210, 220), TextSize = 13, Font = Enum.Font.GothamMedium, TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 2 }, row)
	local onPos = UDim2.new(1, -19, 0.5, -8)
	local offPos = UDim2.new(0, 3, 0.5, -8)
	local bg = N("TextButton", { Size = UDim2.new(0, 42, 0, 22), Position = UDim2.new(1, -42, 0.5, -11), BackgroundColor3 = default and GREEN or GREY, Text = "", AutoButtonColor = false, ZIndex = 2 }, row)
	round(bg)
	local knob = N("Frame", { Size = UDim2.new(0, 16, 0, 16), Position = default and onPos or offPos, BackgroundColor3 = WHITE, ZIndex = 2 }, bg)
	round(knob)
	local st = default
	translatable[#translatable + 1] = function() lbl.Text = T(key) end
	local function apply(v)
		st = v
		Tween:Create(knob, TweenInfo.new(0.28, Enum.EasingStyle.Back, Enum.EasingDirection.Out), { Position = st and onPos or offPos }):Play()
		Tween:Create(bg, TweenInfo.new(0.2), { BackgroundColor3 = st and GREEN or GREY }):Play()
		Tween:Create(lbl, TweenInfo.new(0.12), { TextTransparency = 0.35 }):Play()
		task.delay(0.12, function()
			if lbl and lbl.Parent then
				Tween:Create(lbl, TweenInfo.new(0.22), { TextTransparency = 0 }):Play()
			end
		end)
		if cb then cb(st) end
	end
	bg.MouseButton1Click:Connect(function() apply(not st) end)
	if not UIS.TouchEnabled then
		bg.MouseEnter:Connect(function() Tween:Create(bg, TweenInfo.new(0.15), { BackgroundTransparency = 0.25 }):Play() end)
		bg.MouseLeave:Connect(function() Tween:Create(bg, TweenInfo.new(0.15), { BackgroundTransparency = 0 }):Play() end)
	end
	return apply
end

local function selector(key, options, default, cb)
	order = order + 1
	local row = N("Frame", { Size = UDim2.new(1, 0, 0, 50), BackgroundTransparency = 1, ZIndex = 2, LayoutOrder = order }, scroll)
	local lbl = N("TextLabel", { Size = UDim2.new(1, 0, 0, 15), BackgroundTransparency = 1, Text = T(key), TextColor3 = Color3.fromRGB(210, 210, 220), TextSize = 13, Font = Enum.Font.GothamMedium, TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 2 }, row)
	local btnRow = N("Frame", { Size = UDim2.new(1, 0, 0, 28), Position = UDim2.new(0, 0, 0, 18), BackgroundTransparency = 1, ZIndex = 2 }, row)
	N("UIListLayout", { FillDirection = Enum.FillDirection.Horizontal, HorizontalAlignment = Enum.HorizontalAlignment.Center, Padding = UDim.new(0, 4) }, btnRow)
	local btns = {}
	translatable[#translatable + 1] = function()
		lbl.Text = T(key)
		for nm, b in pairs(btns) do b.Text = T(nm) end
	end
	local function pick(nm)
		for _, v in pairs(btns) do v.BackgroundColor3 = OFF end
		if btns[nm] then btns[nm].BackgroundColor3 = Color3.fromRGB(60, 90, 200) end
		if cb then cb(nm) end
	end
	for _, nm in ipairs(options) do
		local b = N("TextButton", { Size = UDim2.new(0.5, -2, 1, 0), BackgroundColor3 = OFF, Text = T(nm), TextColor3 = WHITE, TextSize = 11, Font = Enum.Font.GothamBold, ZIndex = 2 }, btnRow)
		if nm == default then b.BackgroundColor3 = Color3.fromRGB(60, 90, 200) end
		N("UICorner", { CornerRadius = UDim.new(0, 6) }, b)
		if not UIS.TouchEnabled then
			b.MouseEnter:Connect(function() Tween:Create(b, TweenInfo.new(0.15), { BackgroundTransparency = 0.25 }):Play() end)
			b.MouseLeave:Connect(function() Tween:Create(b, TweenInfo.new(0.15), { BackgroundTransparency = 0 }):Play() end)
		end
		b.MouseButton1Click:Connect(function() pick(nm) end)
		btns[nm] = b
	end
	return pick
end

local function slider(key, cb)
	order = order + 1
	local row = N("Frame", { Size = UDim2.new(1, 0, 0, 40), BackgroundTransparency = 1, ZIndex = 2, LayoutOrder = order }, scroll)
	local lbl = N("TextLabel", { Size = UDim2.new(1, 0, 0, 15), BackgroundTransparency = 1, TextColor3 = Color3.fromRGB(180, 180, 190), TextSize = 12, TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 2 }, row)
	local track = N("Frame", { Size = UDim2.new(1, 0, 0, 6), Position = UDim2.new(0, 0, 0, 25), BackgroundColor3 = OFF, ZIndex = 2 }, row)
	round(track)
	local fill = N("Frame", { BackgroundColor3 = Color3.fromRGB(90, 120, 240), ZIndex = 2 }, track)
	round(fill)
	local knob = N("Frame", { Size = UDim2.new(0, 14, 0, 14), Position = UDim2.new(1, -7, 0.5, -7), BackgroundColor3 = WHITE, ZIndex = 2 }, fill)
	round(knob)
	local cur = 0
	local function set(v)
		v = math.clamp(v, 0, 1)
		cur = v
		fill.Size = UDim2.new(v, 0, 1, 0)
		lbl.Text = T(key) .. ": " .. string.format("%d", math.floor(v * 100))
	end
	translatable[#translatable + 1] = function()
		lbl.Text = T(key) .. ": " .. string.format("%d", math.floor(cur * 100))
	end
	local dragging = false
	local function upd(i)
		local px = math.clamp(i.Position.X - track.AbsolutePosition.X, 0, track.AbsoluteSize.X)
		local pr = px / track.AbsoluteSize.X
		set(pr)
		cb(pr)
	end
	track.InputBegan:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			scroll.ScrollingEnabled = false
			upd(i)
		end
	end)
	UIS.InputChanged:Connect(function(i)
		if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then upd(i) end
	end)
	UIS.InputEnded:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			dragging = false
			scroll.ScrollingEnabled = true
		end
	end)
	return set
end

local function makeLink(pl, ch)
	if not myLow or not myHrp then return end
	local hrp = ch:FindFirstChild("HumanoidRootPart")
	if not hrp then return end
	local foot = getFoot(ch)
	if not foot then return end
	local d = esp[pl]
	if not d then d = {} esp[pl] = d end
	if d.footAtt then d.footAtt:Destroy() end
	if d.low then d.low:Destroy() end
	if d.footStick then d.footStick:Destroy() end
	if d.link then d.link:Destroy() end
	d.footAtt = N("Attachment", { Name = "KyloroxFoot" }, foot)
	d.low = N("Attachment", { Name = "KyloroxLow", Position = Vector3.new(0, -CFG.DROP, 0) }, foot)

local function refresh()
	for _, d in pairs(esp) do
		if d.hl then
			d.hl.FillColor = STYLE.fill.Color
			d.hl.FillTransparency = STYLE.fill.Transparency
			d.hl.OutlineColor = rainbowColor
			d.hl.OutlineTransparency = STYLE.border.Transparency
		end
		if d.name then d.name.Visible = nameOn end
		if d.dist then d.dist.Visible = nameOn end
		if d.barRow then d.barRow.Visible = hpOn end
	end
	if myStick then myStick.Enabled = linkOn end
end

local acc = 0
RunService.Heartbeat:Connect(function(dt)
	if not running or not gui or not gui.Parent then return end
	acc = acc + dt
	if acc < CFG.UPDATE_RATE then return end
	acc = 0

	local cam = workspace.CurrentCamera
	local camPos = Vector3.zero
	if cam then camPos = cam.CFrame.Position end
	local cullDist = CFG.CULL_DIST
	local cullSq = cullDist * cullDist
	local pickScore = nil
	local locked = aimPick
	local lockOk = false
	if locked and cam and locked.Parent then
		local lockHum = locked.Parent:FindFirstChildOfClass("Humanoid")
		local lx = locked.Position.X - camPos.X
		local ly = locked.Position.Y - camPos.Y
		local lz = locked.Position.Z - camPos.Z
		if lockHum and lockHum.Health > 0 and (lx*lx + ly*ly + lz*lz) <= cullSq then
			local sp, onScreen = cam:WorldToViewportPoint(locked.Position)
			if onScreen then
				local sc = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
				local sd = (Vector2.new(sp.X, sp.Y) - sc).Magnitude
				if (not fovOn or sd <= (fovSize / 2)) and (not aimLos or hasLineOfSight(camPos, locked, locked.Parent)) then
					lockOk = true
				end
			end
		end
	end
	aimPick = lockOk and locked or nil

	for pl, d in pairs(esp) do
		if not pl.Parent then
			esp[pl] = nil
		else
			local ch = pl.Character
			local hrp, hum
			if ch then
				hrp = ch:FindFirstChild("HumanoidRootPart")
				hum = ch:FindFirstChildOfClass("Humanoid")
			end

			local inRange = false
			local dist = 0
			local screenDist = nil
			if hrp and cam then
				local targetPart = hrp
				if targetMode == "head" and ch:FindFirstChild("Head") then
					targetPart = ch.Head
				end

				local dx = targetPart.Position.X - camPos.X
				local dy = targetPart.Position.Y - camPos.Y
				local dz = targetPart.Position.Z - camPos.Z
				local distSq = dx*dx + dy*dy + dz*dz

				if distSq <= cullSq then
					inRange = true
					dist = math.floor(math.sqrt(distSq))

					if fovOn or rmbActive then
						local screenPos, onScreen = cam:WorldToViewportPoint(targetPart.Position)
						if onScreen then
							local screenCenter = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
							screenDist = (Vector2.new(screenPos.X, screenPos.Y) - screenCenter).Magnitude
							if fovOn and screenDist > (fovSize / 2) then
								inRange = false
							end
						elseif fovOn then
							inRange = false
						end
					end

					if not lockOk and rmbActive and fovOn and inRange and screenDist and hum and hum.Health > 0
						and (not pickScore or screenDist < pickScore)
						and (not aimLos or hasLineOfSight(camPos, targetPart, ch)) then
						pickScore = screenDist
						aimPick = targetPart
					end
				end
			end

			local wantHl = inRange and espOn
			local wantBeam = inRange and linkOn
			local wantTag = inRange and (nameOn or hpOn)

			if d.hl and d.hl.Enabled ~= wantHl then d.hl.Enabled = wantHl end
			if d.footStick and d.footStick.Enabled ~= wantBeam then d.footStick.Enabled = wantBeam end
			if d.link and d.link.Enabled ~= wantBeam then d.link.Enabled = wantBeam end
			if d.tag and d.tag.Enabled ~= wantTag then d.tag.Enabled = wantTag end

			if inRange and hrp and hum and d.dist then
				if d.lastDist ~= dist then
					d.lastDist = dist
					d.dist.Text = dist .. " studs"
					if dist < 60 then d.dist.TextColor3 = GREEN
					elseif dist < 150 then d.dist.TextColor3 = YELL
					else d.dist.TextColor3 = WHITE end
				end
				local hpInt = math.floor(hum.Health)
				local maxHpInt = math.floor(hum.MaxHealth)
				if d.lastHpInt ~= hpInt or d.lastMaxHpInt ~= maxHpInt then
					d.lastHpInt = hpInt
					d.lastMaxHpInt = maxHpInt
					d.hpTxt.Text = hpInt .. " / " .. maxHpInt
					local p = math.clamp(hum.Health / math.max(hum.MaxHealth, 1), 0, 1)
					local lit = math.ceil(p * 10)
					local col
					if p > 0.5 then col = GREEN elseif p > 0.25 then col = YELL else col = RED end
					d.lastLit = lit
					for i = 1, 10 do
						local s = d.sticks[i]
						if s then
							if i <= lit then s.BackgroundColor3 = col
							else s.BackgroundColor3 = Color3.fromRGB(40, 40, 48) end
						end
					end
				end
			end
		end
	end
end)

local rAcc = 0
RunService.Heartbeat:Connect(function(dt)
	if not running or not gui or not gui.Parent then return end
	rAcc = rAcc + dt
	if rAcc < CFG.RAINBOW_HZ then return end
	hue = (hue + rAcc * CFG.RAINBOW_SPEED) % 1
	rAcc = 0
	rainbowColor = Color3.fromHSV(hue, 1, 1)
	local seq = ColorSequence.new(rainbowColor)
	for _, d in pairs(esp) do
		if d.hl and d.hl.Enabled then d.hl.OutlineColor = rainbowColor end
		if d.link and d.link.Enabled then d.link.Color = seq end
		if d.footStick and d.footStick.Enabled then d.footStick.Color = seq end
	end
	if myStick and myStick.Enabled then myStick.Color = seq end
	if kStroke then kStroke.Color = rainbowColor end
	if kBtn then kBtn.TextColor3 = rainbowColor end
	if bannerStroke then bannerStroke.Color = rainbowColor end
	if fovStroke and fovOn then fovStroke.Color = rainbowColor end
end)

UIS.InputBegan:Connect(function(input, gameProcessed)
	if input.UserInputType == Enum.UserInputType.MouseButton2 then
		if not gameProcessed then rmbActive = true end
	elseif input.UserInputType == Enum.UserInputType.Keyboard and not gameProcessed then
		local ctrl = UIS:IsKeyDown(Enum.KeyCode.LeftControl) or UIS:IsKeyDown(Enum.KeyCode.RightControl)
		if not ctrl then return end
		if input.KeyCode == Enum.KeyCode.H then
			if setFov then setFov(not fovOn) end
		elseif input.KeyCode == Enum.KeyCode.J then
			if setStyle then setStyle(target == "fill" and "border" or "fill") end
		elseif input.KeyCode == Enum.KeyCode.X then
			if togglePanel then togglePanel() end
		end
	end
end)
UIS.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton2 then
		rmbActive = false
		aimPick = nil
	end
end)

pcall(function() RunService:UnbindFromRenderStep("KyloroxAim") end)
RunService:BindToRenderStep("KyloroxAim", Enum.RenderPriority.Camera.Value + 1, function(dt)
	if not running or not gui or not gui.Parent then return end
	if not (rmbActive and fovOn and aimPick) then return end
	local cam = workspace.CurrentCamera
	local part = aimPick
	if not cam or not part or not part.Parent then return end
	local from = cam.CFrame.Position
	local to = part.Position
	if (to - from).Magnitude < 0.1 then return end
	cam.CFrame = cam.CFrame:Lerp(CFrame.lookAt(from, to), 1 - math.exp(-aimSpeed * dt))
end)

		Position = UDim2.new(0.5, -(fovSize/2), 0.5, -(fovSize/2)),

		local cam = workspace.CurrentCamera
		local vp = cam and cam.ViewportSize or Vector2.new(1280, 720)
		local sw = (vp.X - 40) / MAIN_SIZE.X.Offset
		local sh = (vp.Y - 40) / MAIN_SIZE.Y.Offset
		return math.clamp(math.min(sw, sh, 1), 0.6, 1)
	end

	local uiScale = Instance.new("UIScale")
	uiScale.Scale = fitScale()
	uiScale.Parent = main

	local header = N("Frame", { Size = UDim2.new(1, 0, 0, 40), BackgroundTransparency = 1, ZIndex = 2 }, main)
	N("TextLabel", { Size = UDim2.new(1, -108, 1, 0), Position = UDim2.new(0, 15, 0, 0), BackgroundTransparency = 1, Text = "Kylorox", TextColor3 = WHITE, TextSize = 14, Font = Enum.Font.GothamBold, TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 2 }, header)
	local langBtns = {}
	local setLang
	local function makeLangBtn(code, posX)
		local b = N("TextButton", { Size = UDim2.new(0, 28, 0, 26), Position = UDim2.new(1, posX, 0, 7), BackgroundColor3 = Color3.fromRGB(35, 35, 42), Text = string.upper(code), TextColor3 = Color3.fromRGB(220, 220, 230), TextSize = 11, Font = Enum.Font.GothamBold, ZIndex = 3 }, header)
		round(b)
		if not UIS.TouchEnabled then
			b.MouseEnter:Connect(function() Tween:Create(b, TweenInfo.new(0.15), { BackgroundTransparency = 0.2 }):Play() end)
			b.MouseLeave:Connect(function() Tween:Create(b, TweenInfo.new(0.15), { BackgroundTransparency = 0 }):Play() end)
		end
		b.MouseButton1Click:Connect(function() setLang(code) end)
		langBtns[code] = b
	end
	makeLangBtn("es", -97)
	makeLangBtn("en", -65)
	setLang = function(code)
		lang = code
		applyLang()
		for c, b in pairs(langBtns) do
			Tween:Create(b, TweenInfo.new(0.15), { BackgroundColor3 = c == code and Color3.fromRGB(60, 90, 200) or Color3.fromRGB(35, 35, 42) }):Play()
		end
	end
	setLang("es")
	local closeBtn = N("TextButton", { Size = UDim2.new(0, 26, 0, 26), Position = UDim2.new(1, -33, 0, 7), BackgroundColor3 = RED, Text = "X", TextColor3 = WHITE, TextSize = 12, Font = Enum.Font.GothamBold, ZIndex = 2 }, header)
	round(closeBtn)

		closeBtn.MouseEnter:Connect(function() Tween:Create(closeBtn, TweenInfo.new(0.15), { BackgroundTransparency = 0.2 }):Play() end)
		closeBtn.MouseLeave:Connect(function() Tween:Create(closeBtn, TweenInfo.new(0.15), { BackgroundTransparency = 0 }):Play() end)

	toggle("esp", CFG.ESP_ON, function(v) espOn = v refresh() end)
	toggle("name", CFG.NAME_ON, function(v) nameOn = v refresh() end)
	toggle("hp", CFG.HP_ON, function(v) hpOn = v refresh() end)
	toggle("lines", CFG.LINK_ON, function(v) linkOn = v refresh() end)
	setFov = toggle("fov", fovOn, function(v)
		fovOn = v
		if not v then aimPick = nil end
		if fovCircle then fovCircle.Visible = fovOn end
	end)
	toggle("los", CFG.AIM_LOS, function(v) aimLos = v end)

		workspace.CurrentCamera:GetPropertyChangedSignal("ViewportSize"):Connect(function()

		else

		end
